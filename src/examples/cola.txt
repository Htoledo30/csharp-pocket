=== Repetir
--- while | Enquanto a condição for verdadeira.
while (vida > 0)
{
    vida -= 5;
}
--- for | Um número certo de vezes.
for (int i = 1; i <= 10; i++)
{
    Console.WriteLine(i);
}
--- foreach | Cada item de uma lista.
foreach (string item in itens)
{
    Console.WriteLine(item);
}
--- Menu que repete até sair
bool rodando = true;
while (rodando)
{
    Console.WriteLine("[1] Jogar  [2] Sair");
    string opcao = Console.ReadLine();
    if (opcao == "1")
    {
        Console.WriteLine("Jogando...");
    }
    else if (opcao == "2")
    {
        rodando = false;
    }
}

=== Organizar o código
--- Método que devolve um valor

{
    return n * 2;
}
--- Método que só faz algo
void Saudar(string nome)
{
    Console.WriteLine($"Olá, {nome}!");
}
--- Classe @fim | É colocada no fim do arquivo, depois dos comandos.
class Jogador
{
    public string Nome;
    public int Vida = 20;

    public void Ferir(int dano)
    {
        Vida = Math.Clamp(Vida - dano, 0, 20);
    }
}
--- Usar a classe
Jogador heroi = new Jogador();
heroi.Nome = "Teseu";
heroi.Ferir(5);
Console.WriteLine(heroi.Vida);
